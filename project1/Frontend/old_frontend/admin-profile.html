// server.js or routes/admin.js const express = require('express'); const router = express.Router(); const { MongoClient, ObjectId } = require('mongodb'); // MongoDB connection URI const uri = "mongodb://localhost:27017"; // change if hosted const client = new MongoClient(uri); router.get('/admin/profile', async (req, res) => { try { await client.connect(); const db = client.db('CareerMatcherDB'); const usersCollection = db.collection('users'); // Find the admin user const admin = await usersCollection.findOne({ role: 'admin' }); if (!admin) return res.status(404).json({ message: 'Admin not found' }); res.json(admin); } catch (err) { console.error(err); res.status(500).json({ message: 'Server Error' }); } }); module.exports = router;
